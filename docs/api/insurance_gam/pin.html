202    def _compute_linear_predictor(
203        self,
204        x_dict: Dict[str, torch.Tensor],
205        apply_centering: bool = True,
206    ) -> torch.Tensor:
207        """
208        Compute sum_{j<=k} w_{jk} * h_{jk}(x) + b.
209
210        Args:
211            x_dict: Feature tensors on self._device.
212            apply_centering: Subtract pair means (set during fit) for identifiability.
213
214        Returns:
215            Shape (batch_size,).
216        """
217        # Embed all features once
218        embeddings: Dict[str, torch.Tensor] = self.feature_embeddings.embed_all(x_dict)
219
220        pairs = self.interaction_tokens.pair_indices()
221        terms = []
222
223        for pair_idx, (j, k) in enumerate(pairs):
224            fname_j = self.feature_names[j]
225            fname_k = self.feature_names[k]
226
227            phi_j = embeddings[fname_j]
228            phi_k = embeddings[fname_k]
229            token = self.interaction_tokens.get_token(j, k)
230
231            raw = self.shared_net(phi_j, phi_k, token)  # (batch, 1)
232            h = centered_hard_sigmoid(raw).squeeze(-1)   # (batch,)
233
234            w = self.output_weights[pair_idx]
235            term = w * h  # (batch,)
236            terms.append(term)
237
238        # Stack to (batch, n_pairs) then sum over pairs
239        all_terms = torch.stack(terms, dim=1)  # (batch, n_pairs)
240
241        if apply_centering and self._pair_means is not None:
242            all_terms = all_terms - self._pair_means.unsqueeze(0)
243
244        linear_pred = all_terms.sum(dim=1) + self.output_bias.squeeze()
245        return linear_pred

247    def forward(
248        self,
249        x_dict: Dict[str, torch.Tensor],
250        exposure: Optional[torch.Tensor] = None,
251    ) -> torch.Tensor:
252        """
253        Compute predictions f_PIN(x) = exp(linear_predictor) * exposure.
254
255        When exposure is provided, the raw model output is frequency (claims per
256        year) and multiplying by exposure gives expected claim count. For
257        frequency models, typically you'd pass exposure=None and let the caller
258        multiply; this method supports both modes.
259
260        Args:
261            x_dict: Feature tensors. Dict of feature_name -> (batch,) tensor.
262            exposure: Optional per-sample exposure, shape (batch,).
263
264        Returns:
265            Predicted frequency, shape (batch,).
266        """
267        eta = self._compute_linear_predictor(x_dict)
268        # Clamp to avoid exp overflow (GLM link stabilisation)
269        eta = torch.clamp(eta, min=-20.0, max=20.0)
270        mu = torch.exp(eta)
271        if exposure is not None:
272            mu = mu * exposure
273        return mu

339    def fit(
340        self,
341        X_train,
342        y_train: np.ndarray,
343        exposure: Optional[np.ndarray] = None,
344        X_val=None,
345        y_val: Optional[np.ndarray] = None,
346        exposure_val: Optional[np.ndarray] = None,
347        verbose: bool = True,
348    ) -> "PINModel":
349        """
350        Fit the model.
351
352        Args:
353            X_train: Training features. Dict, polars.DataFrame, or pandas.DataFrame.
354            y_train: Observed frequency (claims / exposure), shape (n,).
355            exposure: Per-sample exposure (years at risk), shape (n,).
356            X_val: Validation features (optional). If None, 10% of training data
357                is reserved.
358            y_val: Validation targets.
359            exposure_val: Validation exposure.
360            verbose: Print training progress.
361
362        Returns:
363            self (for chaining).
364        """
365        torch.manual_seed(self.random_seed)
366        np.random.seed(self.random_seed)
367
368        self.to(self._device)
369
370        # Prepare training data
371        x_dict = self._prepare_features(X_train)
372        x_dict = self._to_device_dict(x_dict)
373        y_t = _to_tensor(y_train).to(self._device)
374        exp_t = (
375            _to_tensor(exposure).to(self._device)
376            if exposure is not None
377            else torch.ones_like(y_t)
378        )
379
380        n = y_t.shape[0]
381
382        # Initialise bias to log(mean_frequency) for numerical stability.
383        # This ensures initial predictions are in the right ballpark and
384        # prevents exploding gradients in the first few batches.
385        with torch.no_grad():
386            mean_freq = y_t.mean().clamp(min=1e-8)
387            self.output_bias.fill_(torch.log(mean_freq).item())
388
389        # Build or reserve validation set
390        if X_val is not None:
391            x_val_dict = self._prepare_features(X_val)
392            x_val_dict = self._to_device_dict(x_val_dict)
393            y_val_t = _to_tensor(y_val).to(self._device)
394            exp_val_t = (
395                _to_tensor(exposure_val).to(self._device)
396                if exposure_val is not None
397                else torch.ones_like(y_val_t)
398            )
399        else:
400            # Reserve val_fraction from training
401            val_size = max(1, int(n * self.val_fraction))
402            perm = torch.randperm(n, device=self._device)
403            val_idx = perm[:val_size]
404            train_idx = perm[val_size:]
405
406            x_val_dict = {k: v[val_idx] for k, v in x_dict.items()}
407            y_val_t = y_t[val_idx]
408            exp_val_t = exp_t[val_idx]
409
410            x_dict = {k: v[train_idx] for k, v in x_dict.items()}
411            y_t = y_t[train_idx]
412            exp_t = exp_t[train_idx]
413
414        n_train = y_t.shape[0]
415
416        optimizer = optim.Adam(self.parameters(), lr=self.lr)
417        scheduler = optim.lr_scheduler.ReduceLROnPlateau(
418            optimizer,
419            mode="min",
420            factor=self.lr_factor,
421            patience=self.lr_patience,
422        )
423
424        best_val_loss = float("inf")
425        best_state = None
426        epochs_no_improve = 0
427
428        self.train_history = {"train_loss": [], "val_loss": []}
429
430        for epoch in range(self.max_epochs):
431            self.train()
432            # Shuffle training data
433            perm = torch.randperm(n_train, device=self._device)
434
435            epoch_loss = 0.0
436            n_batches = 0
437
438            for start in range(0, n_train, self.batch_size):
439                end = min(start + self.batch_size, n_train)
440                batch_idx = perm[start:end]
441
442                x_batch = {k: v[batch_idx] for k, v in x_dict.items()}
443                y_batch = y_t[batch_idx]
444                exp_batch = exp_t[batch_idx]
445
446                optimizer.zero_grad()
447                mu = self.forward(x_batch)
448                loss = self._loss_fn(mu, y_batch, exp_batch)
449                if torch.isnan(loss):
450                    continue  # skip NaN batches
451                loss.backward()
452                # Gradient clipping prevents exploding gradients in early training
453                torch.nn.utils.clip_grad_norm_(self.parameters(), max_norm=1.0)
454                # Zero NaN gradients before step
455                for p in self.parameters():
456                    if p.grad is not None and torch.isnan(p.grad).any():
457                        p.grad.zero_()
458                optimizer.step()
459                # Restore any NaN params to their pre-step values
460                for p in self.parameters():
461                    if torch.isnan(p).any():
462                        torch.nan_to_num_(p, nan=0.0)
463
464                epoch_loss += loss.item()
465                n_batches += 1
466
467            train_loss = epoch_loss / n_batches
468
469            # Validation
470            self.eval()
471            with torch.no_grad():
472                mu_val = self.forward(x_val_dict)
473                val_loss = self._loss_fn(mu_val, y_val_t, exp_val_t).item()
474
475            scheduler.step(val_loss)
476
477            self.train_history["train_loss"].append(train_loss)
478            self.train_history["val_loss"].append(val_loss)
479
480            if verbose and (epoch % 50 == 0 or epoch < 10):
481                lr_now = optimizer.param_groups[0]["lr"]
482                print(
483                    f"Epoch {epoch:4d} | train={train_loss:.6f} | val={val_loss:.6f} | lr={lr_now:.2e}"
484                )
485
486            # Early stopping
487            if val_loss < best_val_loss:
488                best_val_loss = val_loss
489                best_state = copy.deepcopy(self.state_dict())
490                epochs_no_improve = 0
491            else:
492                epochs_no_improve += 1
493                if epochs_no_improve >= self.patience:
494                    if verbose:
495                        print(f"Early stopping at epoch {epoch}. Best val={best_val_loss:.6f}")
496                    break
497
498        # Restore best weights
499        if best_state is not None:
500            self.load_state_dict(best_state)
501
502        # Post-hoc centering: compute pair means on training data
503        self._compute_pair_centering(x_dict)
504
505        self._is_fitted = True
506        return self

508    def _compute_pair_centering(
509        self, x_dict: Dict[str, torch.Tensor], batch_size: int = 1024
510    ) -> None:
511        """
512        Compute mean of w_{jk} * h_{jk}(x) over training data for each pair.
513
514        Centering ensures that the bias b absorbs the intercept and the
515        interaction terms have zero mean — making w_{jk} interpretable as
516        the amplitude of the interaction effect.
517
518        Stores result in self._pair_means, shape (n_pairs,).
519        """
520        self.eval()
521        n = x_dict[self.feature_names[0]].shape[0]
522        n_pairs = self.q * (self.q + 1) // 2
523        pairs = self.interaction_tokens.pair_indices()
524
525        pair_sums = torch.zeros(n_pairs, device=self._device)
526        n_seen = 0
527
528        with torch.no_grad():
529            for start in range(0, n, batch_size):
530                end = min(start + batch_size, n)
531                x_batch = {k: v[start:end] for k, v in x_dict.items()}
532                embeddings = self.feature_embeddings.embed_all(x_batch)
533                batch_n = end - start
534
535                for pair_idx, (j, k) in enumerate(pairs):
536                    fname_j = self.feature_names[j]
537                    fname_k = self.feature_names[k]
538                    phi_j = embeddings[fname_j]
539                    phi_k = embeddings[fname_k]
540                    token = self.interaction_tokens.get_token(j, k)
541                    raw = self.shared_net(phi_j, phi_k, token)
542                    h = centered_hard_sigmoid(raw).squeeze(-1)
543                    w = self.output_weights[pair_idx]
544                    pair_sums[pair_idx] += (w * h).sum()
545
546                n_seen += batch_n
547
548        self._pair_means = pair_sums / n_seen

591    def pair_contributions(
592        self,
593        X,
594    ) -> Dict[Tuple[str, str], np.ndarray]:
595        """
596        Compute w_{jk} * h_{jk}(x) for each pair and each sample.

604        Returns:
605            Dict mapping (fname_j, fname_k) -> array of shape (n,).
606        """
607        if not self._is_fitted:
608            raise RuntimeError("Call fit() before pair_contributions().")
609
610        self.eval()
611        x_dict = self._prepare_features(X)
612        x_dict = self._to_device_dict(x_dict)
613        pairs = self.interaction_tokens.pair_indices()
614        result = {}
615
616        with torch.no_grad():
617            embeddings = self.feature_embeddings.embed_all(x_dict)
618
619            for pair_idx, (j, k) in enumerate(pairs):
620                fname_j = self.feature_names[j]
621                fname_k = self.feature_names[k]
622                phi_j = embeddings[fname_j]
623                phi_k = embeddings[fname_k]
624                token = self.interaction_tokens.get_token(j, k)
625                raw = self.shared_net(phi_j, phi_k, token)
626                h = centered_hard_sigmoid(raw).squeeze(-1)
627                w = self.output_weights[pair_idx]
628                contrib = (w * h)
629                if self._pair_means is not None:
630                    contrib = contrib - self._pair_means[pair_idx]
631                result[(fname_j, fname_k)] = contrib.cpu().numpy()
632
633        return result

635    def main_effects(
636        self,
637        X_background,
638        n_grid: int = 100,
639    ) -> Dict[str, Tuple[np.ndarray, np.ndarray]]:
640        """
641        Compute main effect curves for each feature.
642
643        For continuous features: evaluates h_{jj} over an evenly-spaced grid
644        while fixing all other features to background means.
645
646        Returns:
647            Dict mapping feature_name -> (grid_values, effect_values).
648            For categoricals: grid_values are integer category codes.
649        """
650        if not self._is_fitted:
651            raise RuntimeError("Call fit() before main_effects().")
652
653        self.eval()
654        bg_dict = self._prepare_features(X_background)
655        bg_dict = self._to_device_dict(bg_dict)
656
657        result = {}
658
659        with torch.no_grad():
660            for i, fname in enumerate(self.feature_names):
661                spec = self.features[fname]
662
663                if spec == "continuous":
664                    vals = bg_dict[fname].float()
665                    lo, hi = vals.min().item(), vals.max().item()
666                    grid = torch.linspace(lo, hi, n_grid, device=self._device)
667                elif isinstance(spec, int):
668                    grid = torch.arange(spec, device=self._device)
669                else:
670                    continue
671
672                # Build batch: vary feature i, fix others to background mean
673                n_grid_pts = grid.shape[0]
674                x_eval = {}
675                for fname2 in self.feature_names:
676                    spec2 = self.features[fname2]
677                    if fname2 == fname:
